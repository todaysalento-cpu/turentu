// TURENTU POPBUS V2 ƒ?" Preview multimissione isolata (nessuna scrittura DB).
// Collocazione prevista: services/popbus/v2/formation-multimission-preview.js
// Il chiamante fornisce missioni PIXEL giÇÿ validate, veicoli e callback operative.
import { generaCicliDueMissioniShadow } from './pixel-multimission-shadow.js';
import { valutaEconomiaCiclo } from './cycle-pricing.js';

const finite = x => x !== null && x !== undefined && x !== '' && Number.isFinite(Number(x));
const nonnegative = x => finite(x) && Number(x) >= 0;
const idOk = x => Number.isSafeInteger(Number(x)) && Number(x) > 0;
const money = x => Math.round((Number(x) + Number.EPSILON) * 100) / 100;
const fail = msg => { throw new Error(`FORMATION_MULTIMISSION_PREVIEW: ${msg}`); };

function validateMission(m) {
  if (!m || !idOk(m.pixelOpportunitaId) || !Array.isArray(m.requests) || !m.requests.length)
    fail('missione PIXEL non valida');
  if (!finite(m.departure) || !finite(m.arrival) || Number(m.departure) >= Number(m.arrival))
    fail('orari missione non validi');
  if (!nonnegative(m.kmMissione) || !nonnegative(m.peak)) fail('distanza o occupazione non valida');
  if (!m.from || !m.to || ![m.from.lat,m.from.lon,m.to.lat,m.to.lon].every(finite))
    fail('coordinate missione mancanti');
  const ids = m.requests.map(r => r.id);
  if (!ids.every(idOk) || new Set(ids.map(Number)).size !== ids.length ||
      !m.requests.every(r => nonnegative(r.prezzo))) fail('richieste o prezzi non validi');
  if (m.validated !== true) fail('missione priva di attestazione di validazione');
}

function validateVehicle(v) {
  if (!v || !idOk(v.id) || ![v.lat,v.lon,v.euroKm].every(finite) ||
      !nonnegative(v.euroKm) || !Number.isSafeInteger(Number(v.posti_totali)) ||
      Number(v.posti_totali) < 1) fail('veicolo non valido');
}

/**
 * Diagnostica a sola lettura, NON certificazione di attivabilitÇÿ.
 * missions: missioni PIXEL con validated:true (validazione temporale ai nodi,
 * qualitÇÿ, capienza, richieste libere e regole A/R eseguita dal chiamante).
 * validatePair(a,b): controllo obbligatorio dei vincoli incrociati A/R e delle
 * eventuali regole di classe; deve restituire true soltanto se la coppia Çù valida.
 * Le callback movement, isAvailable, hasConflicts devono essere reali e fail-closed.
 * Nessuna selezione TOP-K o filtro di redditivitÇÿ prima del routing.
 */
export async function anteprimaCicliMultimissione({
  missions, vehicles, movement, isAvailable, hasConflicts, validatePair,
  now = Date.now(), maxPairs = 10000, maxRoutingCalls = 100,
  costoAttesaOraEuro = 0, altriCostiEuro = 0, margineMinimoEuro = 0
} = {}) {
  if (!Array.isArray(missions) || !Array.isArray(vehicles) ||
      typeof movement !== 'function' || typeof isAvailable !== 'function' ||
      typeof hasConflicts !== 'function' || typeof validatePair !== 'function')
    fail('dipendenze o validatore di coppia mancanti');
  if (!finite(now) || !Number.isSafeInteger(maxPairs) || maxPairs < 1 ||
      !Number.isSafeInteger(maxRoutingCalls) || maxRoutingCalls < 1 ||
      ![costoAttesaOraEuro, altriCostiEuro, margineMinimoEuro].every(nonnegative))
    fail('parametri di simulazione non validi');
  missions.forEach(validateMission);
  vehicles.forEach(validateVehicle);

  // Il filtro delle coppie Çù indipendente dall'economia. La duplicazione delle
  // richieste Çù verificata anche nel generatore Shadow.
  const admissible = [];
  for (let i = 0; i < missions.length; i++) {
    for (let j = 0; j < missions.length; j++) {
      if (i === j) continue;
      const a = missions[i], b = missions[j];
      if (Number(a.arrival) > Number(b.departure) || Number(a.departure) <= Number(now)) continue;
      const ids = [...a.requests, ...b.requests].map(r => Number(r.id));
      if (new Set(ids).size !== ids.length) continue;
      if (await validatePair(a,b) !== true) continue;
      admissible.push([a,b]);
    }
  }

  // Shadow conta coppie per veicolo; falliamo prima di eseguire routing.
  const possible = admissible.reduce((count,[a,b]) => count + vehicles.filter(v =>
    Number(a.peak) <= Number(v.posti_totali) &&
    Number(b.peak) <= Number(v.posti_totali)).length, 0);
  if (possible > maxPairs) fail(`limite coppie superato (${possible} > ${maxPairs}); nessun risultato parziale`);

  const routingCache = new Map();
  let routingCalls = 0;
  const cachedMovement = async (a,b) => {
    const key = `${Number(a.lat)},${Number(a.lon)}>${Number(b.lat)},${Number(b.lon)}`;
    if (routingCache.has(key)) return routingCache.get(key);
    if (++routingCalls > maxRoutingCalls) fail('budget routing esaurito; nessun risultato parziale');
    const result = await movement(a,b);
    if (!result || ![result.metres,result.seconds].every(nonnegative)) fail('routing non valido');
    routingCache.set(key,result);
    return result;
  };
  const cicli = [];
  let attempted = 0;
  for (const [a,b] of admissible) {
    // Due missioni sono passate in ordine; Shadow valuta solo a -> b.
    const out = await generaCicliDueMissioniShadow({
      missions:[a,b], vehicles, movement:cachedMovement, isAvailable,
      hasConflicts, now:Number(now), maxPairs:Math.max(1,vehicles.length),
      maxRoutingCalls:Math.max(1,maxRoutingCalls), costoAttesaOraEuro
    });
    attempted += out.attempted;
    for (const c of out.cicli) {
      const v = vehicles.find(x => Number(x.id) === Number(c.veicolo));
      if (!v) fail('veicolo risultato non trovato');
      const s = c._shadow;
      const economy = valutaEconomiaCiclo({
        ricaviRichiesteEuro:[...a.requests,...b.requests].map(r => Number(r.prezzo)),
        kmAvvicinamento:Number(s.approach.metres)/1000,
        kmMissioni:Number(a.kmMissione)+Number(b.kmMissione),
        kmTrasferimenti:Number(s.transfer.metres)/1000,
        kmRitornoBase:Number(s.returnTrip.metres)/1000,
        euroKm:Number(v.euroKm), oreAttesa:Number(s.waitingHours),
        costoAttesaOraEuro:Number(costoAttesaOraEuro),
        altriCostiEuro:Number(altriCostiEuro), margineMinimoEuro:Number(margineMinimoEuro)
      });
      cicli.push({
        tipo_ciclo:'due_missioni', simulazione_non_prenotabile:true,
        attendibilita:'diagnostica', veicolo_id:Number(v.id),
        pixel_opportunita_ids:[Number(a.pixelOpportunitaId),Number(b.pixelOpportunitaId)],
        richiesta_ids:[...a.requests,...b.requests].map(r => Number(r.id)),
        partenza_base:new Date(c.partenza_base).toISOString(),
        ritorno_base:new Date(c.ritorno_base).toISOString(),
        ricavo_lordo_euro:economy.ricavoLordoEuro,
        costo_completo_euro:economy.costoCompletoEuro,
        soglia_attivazione_lorda_euro:economy.sogliaAttivazioneLordaEuro,
        margine_euro:economy.margineEuro,
        deficit_lordo_euro:money(Math.max(0,economy.sogliaAttivazioneLordaEuro-economy.ricavoLordoEuro)),
        soglia_raggiunta_economicamente:economy.economicamenteAttivabile,
        km_avvicinamento:Number(s.approach.metres)/1000,
        km_missioni:Number(a.kmMissione)+Number(b.kmMissione),
        km_trasferimento:Number(s.transfer.metres)/1000,
        km_ritorno:Number(s.returnTrip.metres)/1000,
        ore_attesa:Number(s.waitingHours)
      });
    }
  }
  cicli.sort((a,b) => a.deficit_lordo_euro-b.deficit_lordo_euro ||
    b.margine_euro-a.margine_euro || a.veicolo_id-b.veicolo_id);
  return {
    shadow:true, dbWrites:0, simulazione_non_prenotabile:true,
    completezza:'limitata_alle_missioni_fornite',
    coppie_ammesse:admissible.length, coppie_valutate:attempted,
    routing_calls:routingCalls, cicli, migliore:cicli[0] ?? null
  };
}
// TURENTU POPBUS V2 ƒ?" Ottimizzazione temporale isolata, senza I/O e senza scritture.
// Collocare in services/popbus/v2/pixel-multimission-time-optimizer.js
const finite = v => v !== null && v !== undefined && Number.isFinite(Number(v));
const fail = s => { throw new Error(`PIXEL_TIME_OPTIMIZER: ${s}`); };

/**
 * Le finestre lower/upper DEVONO essere state calcolate dal chiamante come
 * intersezione della finestra PIXEL e delle tolleranze di tutte le richieste
 * ai rispettivi nodi di salita. Il modulo non crea direttrici o richieste.
 * movement restituisce il trasferimento stradale reale {metres,seconds}.
 * La funzione massimizza il margine temporale (A al piÇû presto, B al piÇû tardi),
 * non il profitto nÇ¸ la preferenza degli utenti dentro la finestra.
 */
export async function ottimizzaCoppiaTemporale({a,b,movement}={}) {
  if (!a || !b || typeof movement !== 'function') fail('argomenti mancanti');
  for (const [name,m] of [['A',a],['B',b]]) {
    if (![m.lower,m.upper,m.departure,m.arrival].every(finite)) fail(`orari ${name} non validi`);
    if (Number(m.lower)>Number(m.upper) || Number(m.departure)>=Number(m.arrival)) fail(`finestra ${name} non valida`);
    if (Number(m.departure)<Number(m.lower)-1 || Number(m.departure)>Number(m.upper)+1)
      fail(`partenza originale ${name} fuori finestra`);
    if (!m.from || !m.to || ![m.from.lat,m.from.lon,m.to.lat,m.to.lon].every(finite))
      fail(`coordinate ${name} non valide`);
    if (!Array.isArray(m.requests) || !m.requests.length) fail(`richieste ${name} mancanti`);
  }
  const ids=[...a.requests,...b.requests].map(r=>Number(r.id));
  if (ids.some(id=>!Number.isSafeInteger(id)||id<=0) || new Set(ids).size!==ids.length)
    return {stato:'RICHIESTE_DUPLICATE',fattibile:false};
  const transfer=await movement(a.to,b.from);
  if (!transfer || ![transfer.metres,transfer.seconds].every(finite) ||
      Number(transfer.metres)<0 || Number(transfer.seconds)<0) fail('routing non valido');
  const duration=Number(a.arrival)-Number(a.departure);
  const transferMs=Number(transfer.seconds)*1000;
  const originalMargin=Number(b.departure)-Number(a.arrival)-transferMs;
  const optimizedMargin=Number(b.upper)-Number(a.lower)-duration-transferMs;
  const result={
    stato: optimizedMargin < 0 ? 'TRASFERIMENTO_IMPOSSIBILE' :
      originalMargin < 0 ? 'RECUPERATA' : 'GIA_FATTIBILE',
    fattibile:optimizedMargin>=0,
    recuperata:originalMargin<0 && optimizedMargin>=0,
    margine_originale_min:Number((originalMargin/60000).toFixed(3)),
    margine_ottimizzato_min:Number((optimizedMargin/60000).toFixed(3)),
    trasferimento_km:Number(transfer.metres)/1000,
    trasferimento_min:Number(transfer.seconds)/60,
    partenza_a:Number(a.lower),partenza_b:Number(b.upper),
    missione_a:{...a,departure:Number(a.lower),arrival:Number(a.lower)+duration},
    missione_b:{...b,departure:Number(b.upper),arrival:Number(b.upper)+(Number(b.arrival)-Number(b.departure))}
  };
  return result;
}
// TURENTU POPBUS V2 ƒ?" pianificatore economico-temporale ISOLATO, puro salvo callback routing.
// Non legge/scrive DB, non modifica PIXEL, non attiva corse.
const valid = x => x !== null && x !== undefined && x !== '' && Number.isFinite(Number(x));
const error = s => { throw new Error(`PIXEL_ECONOMIC_TIME_OPTIMIZER: ${s}`); };
const round = x => Math.round((x + Number.EPSILON) * 1000) / 1000;

/**
 * Missioni gia' validate dal chiamante. lower/upper: intersezione delle finestre
 * PIXEL e delle tolleranze delle richieste ai rispettivi nodi di salita (epoch ms).
 * Durate di missione e routing sono costanti rispetto alla partenza.
 * Questo modulo ottimizza SOLO il tempo di attesa fra DUE missioni; non modifica
 * prezzo, km, richieste o direttrici. Non certifica qualita', turni, conflitti, A/R.
 * validateShiftedMission deve riverificare ai nodi di salita la missione spostata.
 */
export async function ottimizzaAttesaDueMissioni({
  a,b,movement,validateShiftedMission,costoAttesaOraEuro=0
}={}) {
  if (!a || !b || typeof movement !== 'function' || typeof validateShiftedMission !== 'function') error('dipendenze mancanti');
  if (!valid(costoAttesaOraEuro) || Number(costoAttesaOraEuro)<0) error('costo attesa non valido');
  for (const [name,m] of [['A',a],['B',b]]) {
    if (![m.lower,m.upper,m.departure,m.arrival].every(valid) ||
      Number(m.lower)>Number(m.upper) || Number(m.departure)>=Number(m.arrival) ||
      Number(m.departure)<Number(m.lower) || Number(m.departure)>Number(m.upper)) error(`orari ${name} non validi`);
    if (!m.from || !m.to || ![m.from.lat,m.from.lon,m.to.lat,m.to.lon].every(valid)) error(`coordinate ${name} non valide`);
    if (!Array.isArray(m.requests) || !m.requests.length) error(`richieste ${name} mancanti`);
  }
  const ids=[...a.requests,...b.requests].map(r=>Number(r.id));
  if (ids.some(x=>!Number.isSafeInteger(x)||x<=0) || new Set(ids).size!==ids.length)
    return {stato:'RICHIESTE_DUPLICATE',fattibile:false};
  const transfer=await movement(a.to,b.from);
  if (!transfer || ![transfer.metres,transfer.seconds].every(valid) ||
    Number(transfer.metres)<0 || Number(transfer.seconds)<0) error('routing non valido');
  const durationA=Number(a.arrival)-Number(a.departure);
  const durationB=Number(b.arrival)-Number(b.departure);
  const transferMs=Number(transfer.seconds)*1000;
  const latestA=Math.min(Number(a.upper),Number(b.upper)-durationA-transferMs);
  if (latestA<Number(a.lower)) return {
    stato:'TRASFERIMENTO_IMPOSSIBILE',fattibile:false,
    deficit_min:round((Number(a.lower)+durationA+transferMs-Number(b.upper))/60000),
    trasferimento_km:round(Number(transfer.metres)/1000)
  };
  // A parte il piu' tardi possibile; B parte non appena il veicolo arriva,
  // salvo che la sua finestra inizi dopo. E' un ottimo del costo di attesa
  // con tariffa non negativa e durata di viaggio costante.
  const depA=latestA;
  const depB=Math.max(Number(b.lower),depA+durationA+transferMs);
  const waitMs=Math.max(0,depB-depA-durationA-transferMs);
  const shiftedA={...a,departure:depA,arrival:depA+durationA};
  const shiftedB={...b,departure:depB,arrival:depB+durationB};
  if (await validateShiftedMission(shiftedA,a)!==true ||
      await validateShiftedMission(shiftedB,b)!==true)
    return {stato:'VALIDAZIONE_NODI_FALLITA',fattibile:false};
  const originalWaitMs=Number(b.departure)-Number(a.arrival)-transferMs;
  return {
    stato:'FATTIBILE_TEMPORALMENTE',fattibile:true,
    recuperata:originalWaitMs<0,
    missione_a:shiftedA,missione_b:shiftedB,
    partenza_a:new Date(depA).toISOString(),partenza_b:new Date(depB).toISOString(),
    trasferimento_km:round(Number(transfer.metres)/1000),
    attesa_min:round(waitMs/60000),
    costo_attesa_euro:round(waitMs/3600000*Number(costoAttesaOraEuro)),
    margine_temporale_residuo_min:round((Number(b.upper)-depB)/60000),
    note:'Ottimo solo per attesa fra due missioni, non per ciclo globale o preferenze passeggeri'
  };
}
// TURENTU POPBUS V2 ƒ?" GENERATORE MULTI-MISSIONE SHADOW (modulo puro)
// Salva come services/popbus/v2/pixel-multimission-shadow.js
// Nessun database, commit, modifica a PIXEL o attivazione.
// Il chiamante deve fornire missioni PIXEL gia' validate, routing verificato,
// prezzi originali, disponibilita e conflitti. NON e' un certificatore produttivo.
import { valutaEconomiaCiclo } from './cycle-pricing.js';

const fail = message => { throw new Error(`PIXEL_MULTIMISSION_SHADOW: ${message}`); };
const num = (x, name) => { const n=Number(x); if (!Number.isFinite(n) || n<0) fail(`${name} non valido`); return n; };
const positiveId = x => Number.isSafeInteger(Number(x)) && Number(x)>0;
function normalizeMission(m) {
  if (!m || !positiveId(m.pixelOpportunitaId)) fail('opportunita PIXEL non valida');
  if (!m.from || !m.to || !Number.isFinite(Number(m.from.lat)) || !Number.isFinite(Number(m.from.lon)) || !Number.isFinite(Number(m.to.lat)) || !Number.isFinite(Number(m.to.lon))) fail('estremi missione mancanti');
  if (!Array.isArray(m.requests) || !m.requests.length) fail('richieste mancanti');
  const ids=m.requests.map(r=>Number(r.id));
  if (!ids.every(positiveId) || new Set(ids).size!==ids.length) fail('ID richieste non validi');
  const prices=m.requests.map(r=>num(r.prezzo,'prezzo richiesta'));
  const start=Number(m.departure), end=Number(m.arrival);
  if (!Number.isFinite(start) || !Number.isFinite(end) || start>=end) fail('orari missione non validi');
  const km=num(m.kmMissione,'kmMissione');
  return {...m, departure:start,arrival:end,kmMissione:km,requestIds:ids,prices};
}
function legCheck(leg, name) {
  if (!leg || !Number.isFinite(Number(leg.metres)) || !Number.isFinite(Number(leg.seconds))) fail(`${name}: routing mancante`);
  return {metres:num(leg.metres,`${name}.metres`),seconds:num(leg.seconds,`${name}.seconds`)};
}
/**
 * Costruisce solo sequenze di 2 missioni dello stesso veicolo.
 * `missions`: [{pixelOpportunitaId,from,to,departure,arrival,kmMissione,
 *               requests:[{id,prezzo}],peak}]. Orari in millisecondi.
 * `vehicles`: [{id,lat,lon,euroKm,posti_totali}].
 * `movement(from,to)` restituisce {metres,seconds} stradali verificati.
 * `isAvailable(vehicle,busyStart,busyEnd)` verifica l'intero ciclo.
 * `hasConflicts(vehicle,busyStart,busyEnd)` verifica l'intero ciclo.
 * Limiti: fallimento esplicito, mai troncamento silenzioso.
 * Non applica da solo vincoli A/R, qualita', finestre ai punti intermedi,
 * costi reali di attesa, o soglie di classe: responsabilita del chiamante.
 */
export async function generaCicliDueMissioniShadow({missions,vehicles,movement,isAvailable,hasConflicts,
  now=Date.now(),maxPairs=10000,maxRoutingCalls=100,costoAttesaOraEuro=0}={}) {
  if (!Array.isArray(missions) || !Array.isArray(vehicles) ||
      typeof movement!=='function' || typeof isAvailable!=='function' || typeof hasConflicts!=='function') fail('dipendenze mancanti');
  if (!Number.isSafeInteger(maxPairs) || maxPairs<1) fail('maxPairs non valido');
  if (!Number.isSafeInteger(maxRoutingCalls) || maxRoutingCalls<1) fail('maxRoutingCalls non valido');
  num(costoAttesaOraEuro,'costoAttesaOraEuro');
  const normalized=missions.map(normalizeMission);
  const results=[]; let attempted=0;
  // Cache per coordinate e direzione: nessuna distanza simmetrica presunta.
  // Non conserva promesse fallite; errori di routing sempre espliciti.
  const routingCache=new Map();
  const routingStats={richieste:0,chiamateEffettive:0,cacheHit:0};
  const keyPoint=p=>`${Number(p.lat)},${Number(p.lon)}`;
  async function cachedMovement(from,to,name) {
    routingStats.richieste++;
    const key=`${keyPoint(from)}>${keyPoint(to)}`;
    if (routingCache.has(key)) {
      routingStats.cacheHit++;
      return routingCache.get(key);
    }
    if (routingStats.chiamateEffettive >= maxRoutingCalls) {
      fail(`limite routing raggiunto (${maxRoutingCalls}): nessun risultato parziale; ${JSON.stringify(routingStats)}`);
    }
    routingStats.chiamateEffettive++;
    const route=legCheck(await movement(from,to),name);
    routingCache.set(key,route);
    return route;
  }
  // Controllo preventivo del limite: nessun routing prima del fail-closed.
  // Stessi filtri e stesso ordine del generatore originale.
  let pairCount=0;
  for (const v of vehicles) {
    if (!positiveId(v.id) || !Number.isFinite(Number(v.lat)) || !Number.isFinite(Number(v.lon))) fail('veicolo non valido');
    const capacity=num(v.posti_totali,'posti_totali');
    num(v.euroKm,'euroKm');
    for (const a of normalized) for (const b of normalized) {
      if (a===b || a.arrival>b.departure || a.departure<=now) continue;
      if (num(a.peak,'peak')>capacity || num(b.peak,'peak')>capacity) continue;
      if (a.requestIds.some(id=>b.requestIds.includes(id))) continue;
      if (++pairCount>maxPairs) fail('limite coppie raggiunto: nessun risultato parziale');
    }
  }
  for (const v of vehicles) {
    if (!positiveId(v.id) || !Number.isFinite(Number(v.lat)) || !Number.isFinite(Number(v.lon))) fail('veicolo non valido');
    const base={lat:Number(v.lat),lon:Number(v.lon)};
    const capacity=num(v.posti_totali,'posti_totali');
    const euroKm=num(v.euroKm,'euroKm');
    for (const a of normalized) for (const b of normalized) {
      if (a===b || a.arrival>b.departure || a.departure<=now) continue;
      if (num(a.peak,'peak')>capacity || num(b.peak,'peak')>capacity) continue;
      if (a.requestIds.some(id=>b.requestIds.includes(id))) continue;
      if (++attempted>maxPairs) fail('limite coppie raggiunto: nessun risultato parziale');
      const approach=await cachedMovement(base,a.from,'avvicinamento');
      const transfer=await cachedMovement(a.to,b.from,'trasferimento');
      const ret=await cachedMovement(b.to,base,'ritorno');
      if (a.arrival+transfer.seconds*1000>b.departure) continue;
      const busyStart=a.departure-approach.seconds*1000;
      const busyEnd=b.arrival+ret.seconds*1000;
      if (busyStart<=now || !(await isAvailable(v,busyStart,busyEnd)) ||
          (await hasConflicts(v,busyStart,busyEnd))) continue;
      const waitingHours=(b.departure-a.arrival-transfer.seconds*1000)/3600000;
      const economia=valutaEconomiaCiclo({
        ricaviRichiesteEuro:[...a.prices,...b.prices],
        kmAvvicinamento:approach.metres/1000,
        kmMissioni:a.kmMissione+b.kmMissione,
        kmTrasferimenti:transfer.metres/1000,
        kmRitornoBase:ret.metres/1000,
        euroKm,oreAttesa:waitingHours,costoAttesaOraEuro
      });
      results.push({veicolo:Number(v.id),richieste:[...a.requestIds,...b.requestIds],
        partenza_base:busyStart,ritorno_base:busyEnd,
        margine_ciclo_con_attesa_euro:economia.margineEuro,
        _economiaRicalcolata:true,
        _pixelOpportunitaIds:[Number(a.pixelOpportunitaId),Number(b.pixelOpportunitaId)],
        _shadow:{missions:[a,b],approach,transfer,returnTrip:ret,economia,waitingHours}});
    }
  }
  return {shadow:true,dbWrites:0,attempted,cicli:results,routingStats};
}
